아래 Json 정책 파일 생성
pod_policy

{
  "Version": "2012-10-17",
  "Statement": [
    {
      "Sid": "AllowDynamoDBPutItemAndKMSDecrypt",
      "Effect": "Allow",
      "Action": [
        "dynamodb:PutItem",
        "kms:Decrypt"
      ],
      "Resource": "*"
    }
  ]
}

aws iam create-policy \
    --policy-name wskorea26-pod-policy \
    --policy-document file://pod_policy.json

eksctl create iamserviceaccount \
  --cluster=$CLUSTER_NAME \
  --namespace=wskorea26 \
  --name=wskorea26-sa \
  --role-name=wskorea26-pod-role \
  --attach-policy-arn=arn:aws:iam::$AWS_ACCOUNT_ID:policy/wskorea26-pod-policy \
  --approve
